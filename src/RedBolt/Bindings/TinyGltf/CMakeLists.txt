set(REDBOLT_TINYGLTF_BINDING_CXX_COMPILER "${REDBOLT_BINDING_CXX_COMPILER}" CACHE FILEPATH
    "C++ compiler used to discover headers for TinyGLTF binding generation" FORCE)

redbolt_add_native_binding_generation(
    TARGET redbolt-generate-tinygltf-bindings
    SURFACE tinygltf
    HEADER "${REDBOLT_TINYGLTF_PARSE_HEADER}"
    HEADER_INCLUDE_PATH "TinyGltfAll.hpp"
    FILE_STEM "${REDBOLT_TINYGLTF_BINDINGS_STEM}"
    OUTPUT_DIR "${REDBOLT_TINYGLTF_GENERATED_DIR}"
    OUTPUT_METADATA "${REDBOLT_TINYGLTF_BINDINGS_JSON}"
    OUTPUT_CS "${REDBOLT_TINYGLTF_BINDINGS_CS}"
    OUTPUT_CPP "${REDBOLT_TINYGLTF_BINDINGS_CPP}"
    NATIVE_LIBRARY_NAME "${REDBOLT_TINYGLTF_NATIVE_LIBRARY_NAME}"
    COMPILER "${REDBOLT_TINYGLTF_BINDING_CXX_COMPILER}"
    INCLUDE_DIRS "${REDBOLT_TINYGLTF_INCLUDE_DIR}"
    DEFINES TINYGLTF_NO_STB_IMAGE TINYGLTF_NO_STB_IMAGE_WRITE
    OWNED_HEADERS "${REDBOLT_TINYGLTF_PARSE_HEADER}" "${REDBOLT_TINYGLTF_INCLUDE_DIR}/tiny_gltf.h" "${REDBOLT_TINYGLTF_INCLUDE_DIR}/json.hpp")
