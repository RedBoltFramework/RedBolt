cmake_minimum_required(VERSION 3.28)

project(RedBoltFramework VERSION 0.1.0 LANGUAGES C CXX)

if(MINGW)
    include_directories(BEFORE SYSTEM "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Toolchains/MinGW/include")
endif()

add_custom_target(redbolt-msbuild-build-guard ALL
    COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/CheckMsBuildBuild.cmake"
    VERBATIM)

option(REDBOLT_WEB_BUILD "Build the framework-free WebAssembly target only" OFF)
option(REDBOLT_ANDROID_BUILD "Build the Android runtime and sample support" OFF)

if(REDBOLT_WEB_BUILD OR EMSCRIPTEN)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build shared libraries" FORCE)
    set(REDBOLT_EXTERNAL_LIBRARY_TYPE STATIC)
else()
    set(BUILD_SHARED_LIBS ON CACHE BOOL "Build shared libraries" FORCE)
    set(REDBOLT_EXTERNAL_LIBRARY_TYPE SHARED)
endif()

if(MSVC)
    add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/std:c++20>")
else()
    add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-std=c++20>")
endif()
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

if(EMSCRIPTEN)
    add_compile_options(-mbulk-memory)
    set_property(GLOBAL PROPERTY RULE_LAUNCH_LINK "${_redbolt_emscripten_environment}")
endif()

if(ANDROID)
    # Keep generated managed signatures identical across Android ABIs. Clang
    # otherwise treats plain char as unsigned on arm64 and signed on x86_64.
    add_compile_options(-fsigned-char)
    add_link_options("-Wl,-z,max-page-size=16384")
endif()

set(REDBOLT_DEPENDENCY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/extern" CACHE PATH "Path to RedBolt dependency sources" FORCE)
set(REDBOLT_SHARPCPP_ROOT "${REDBOLT_DEPENDENCY_ROOT}/sharpcpp" CACHE PATH "Path to the SharpCpp source tree" FORCE)
if(ANDROID)
    if(ANDROID_ABI STREQUAL "arm64-v8a")
        set(_redbolt_default_runtime_identifier android-arm64)
    elseif(ANDROID_ABI STREQUAL "x86_64")
        set(_redbolt_default_runtime_identifier android-x64)
    else()
        message(FATAL_ERROR "Unsupported RedBolt Android ABI: ${ANDROID_ABI}")
    endif()
elseif(WIN32)
    set(_redbolt_default_runtime_identifier win-x64)
else()
    set(_redbolt_default_runtime_identifier linux-x64)
endif()
set(REDBOLT_RUNTIME_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build/bin/$<CONFIG>/${_redbolt_default_runtime_identifier}/extern/runtimes/${_redbolt_default_runtime_identifier}/native" CACHE PATH "RedBolt runtime output directory")
set(REDBOLT_PDB_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build/bin/$<CONFIG>/${_redbolt_default_runtime_identifier}/extern/pdb" CACHE PATH "RedBolt PDB output directory")
set(REDBOLT_GENERATED_ROOT "${CMAKE_BINARY_DIR}/generated" CACHE INTERNAL "RedBolt generated file root")
set(REDBOLT_NATIVE_GENERATED_DIR "${REDBOLT_GENERATED_ROOT}/redbolt-native" CACHE INTERNAL "RedBolt generated native binding root")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/RedBoltNativeModules.cmake")

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/$<CONFIG>")
set(CMAKE_COMPILE_PDB_OUTPUT_DIRECTORY "${REDBOLT_PDB_OUTPUT_DIR}")
set(CMAKE_PDB_OUTPUT_DIRECTORY "${REDBOLT_PDB_OUTPUT_DIR}")
set(CMAKE_BUILD_RPATH "$ORIGIN")
set(CMAKE_INSTALL_RPATH "$ORIGIN")
if(NOT EMSCRIPTEN AND REDBOLT_CONFIGURATION)
    if(CMAKE_CONFIGURATION_TYPES)
        foreach(_redbolt_configuration ${CMAKE_CONFIGURATION_TYPES})
            string(TOUPPER "${_redbolt_configuration}" _redbolt_configuration_upper)
            set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${_redbolt_configuration_upper} "${REDBOLT_RUNTIME_OUTPUT_DIR}")
            set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${_redbolt_configuration_upper} "${REDBOLT_RUNTIME_OUTPUT_DIR}")
        endforeach()
    else()
        set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${REDBOLT_RUNTIME_OUTPUT_DIR}")
        set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${REDBOLT_RUNTIME_OUTPUT_DIR}")
    endif()
endif()

add_subdirectory(cmake)
add_subdirectory(extern)
add_subdirectory(src)
